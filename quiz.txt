import React, { useState } from "react";

function App() {
  const questions = [
    {
      question: "Which language is used with React?",
      options: ["Python", "JavaScript", "C", "SQL"],
      answer: "JavaScript"
    },
    {
      question: "Which hook manages state in React?",
      options: ["useState", "useRoute", "usePage", "useStyle"],
      answer: "useState"
    },
    {
      question: "Which command starts a Vite React app?",
      options: ["npm start", "npm run dev", "git start", "node app"],
      answer: "npm run dev"
    }
  ];

  const [index, setIndex] = useState(0);
  const [score, setScore] = useState(0);
  const [selected, setSelected] = useState("");
  const [finished, setFinished] = useState(false);

  function nextQuestion() {
    if (selected === "") return;

    if (selected === questions[index].answer) {
      setScore(score + 1);
    }

    if (index === questions.length - 1) {
      setFinished(true);
    } else {
      setIndex(index + 1);
      setSelected("");
    }
  }

  return (
    <div style={{ textAlign: "center", margin: "50px" }}>
      <h1>React Quiz App</h1>

      {finished ? (
        <h2>
          Quiz Completed! Score: {score} / {questions.length}
        </h2>
      ) : (
        <>
          <h3>{questions[index].question}</h3>

          {questions[index].options.map((option) => (
            <div key={option}>
              <label>
                <input
                  type="radio"
                  name="answer"
                  value={option}
                  checked={selected === option}
                  onChange={() => setSelected(option)}
                />
                {option}
              </label>
            </div>
          ))}

          <br />

          <button onClick={nextQuestion}>
            {index === questions.length - 1 ? "Finish" : "Next"}
          </button>
        </>
      )}
    </div>
  );
}

export default App;
