import React, { useState } from "react";

function App() {
  const [celsius, setCelsius] = useState("");
  const [fahrenheit, setFahrenheit] = useState("");

  function toFahrenheit(value) {
    setCelsius(value);
    setFahrenheit(value === "" ? "" : (Number(value) * 9 / 5 + 32).toFixed(2));
  }

  function toCelsius(value) {
    setFahrenheit(value);
    setCelsius(value === "" ? "" : ((Number(value) - 32) * 5 / 9).toFixed(2));
  }

  return (
    <div style={{ textAlign: "center", margin: "50px" }}>
      <h1>Temperature Converter</h1>

      <p>Celsius to Fahrenheit</p>
      <input
        type="number"
        value={celsius}
        onChange={(e) => toFahrenheit(e.target.value)}
        placeholder="Enter Celsius"
      />

      <p>Fahrenheit to Celsius</p>
      <input
        type="number"
        value={fahrenheit}
        onChange={(e) => toCelsius(e.target.value)}
        placeholder="Enter Fahrenheit"
      />

      <h3>Celsius: {celsius} °C</h3>
      <h3>Fahrenheit: {fahrenheit} °F</h3>
    </div>
  );
}

export default App;
