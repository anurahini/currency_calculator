import React, { useState } from "react";

function App() {
  const [search, setSearch] = useState("");
  const [recipes, setRecipes] = useState([]);
  const [error, setError] = useState("");

  async function findRecipes() {
    if (!search.trim()) return;

    setError("");
    setRecipes([]);

    try {
      const response = await fetch(
        `https://www.themealdb.com/api/json/v1/1/search.php?s=${encodeURIComponent(search)}`
      );

      const data = await response.json();

      if (data.meals) {
        setRecipes(data.meals);
      } else {
        setError("No recipes found");
      }
    } catch {
      setError("Unable to load recipes. Check your internet.");
    }
  }

  return (
    <div style={{ textAlign: "center", margin: "30px" }}>
      <h1>Recipe Finder</h1>

      <input
        value={search}
        onChange={(e) => setSearch(e.target.value)}
        placeholder="Enter recipe name"
      />

      <button onClick={findRecipes}>Search</button>

      <p>{error}</p>

      {recipes.map((r) => (
        <div
          key={r.idMeal}
          style={{ border: "1px solid gray", margin: "20px auto", padding: "15px", maxWidth: "400px" }}
        >
          <h2>{r.strMeal}</h2>
          <img src={r.strMealThumb} alt={r.strMeal} width="200" />
          <p>Category: {r.strCategory}</p>
          <p>Area: {r.strArea}</p>
          <a href={r.strSource || r.strYoutube} target="_blank" rel="noreferrer">
            View Recipe
          </a>
        </div>
      ))}
    </div>
  );
}

export default App;
